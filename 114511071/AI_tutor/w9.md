## 5. Verification

**My final program:**

```cpp
#include <iostream>
#include <unordered_map>
#include <vector>

using namespace std;

vector<int> firstRepeatedNumber(const vector<int>& nums) {
    unordered_map<int, int> firstIndex;

    for (int i = 0; i < nums.size(); i++) {
        if (firstIndex.count(nums[i])) {
            return {firstIndex[nums[i]], i};
        }

        firstIndex[nums[i]] = i;
    }

    return {};
}

int main() {
    vector<int> nums = {4, 2, 7, 2, 9};
    vector<int> answer = firstRepeatedNumber(nums);

    if (answer.empty()) {
        cout << "No repeated number" << endl;
    } else {
        cout << answer[0] << " " << answer[1] << endl;
    }

    return 0;
}
```
