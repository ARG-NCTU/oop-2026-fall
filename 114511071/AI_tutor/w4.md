### 5. Verification

**My final program:**

#### `delivery.py`

```python
def delivery_fee(weight):
    if weight <= 0 or weight > 30:
        raise ValueError("weight must be between 0 and 30")

    if weight <= 1:
        return 60
    elif weight <= 5:
        return 100
    elif weight <= 20:
        return 180
    else:
        return 300


def average_fee(weights):
    if len(weights) == 0:
        raise ValueError("weights must not be empty")

    total = 0

    for weight in weights:
        total += delivery_fee(weight)

    return total / len(weights)
```

#### `test_delivery.py`

```python
import pytest

from delivery import average_fee, delivery_fee


@pytest.mark.parametrize(
    "weight, expected",
    [
        (0.5, 60),
        (1, 60),
        (1.1, 100),
        (5, 100),
        (5.1, 180),
        (20, 180),
        (20.1, 300),
        (30, 300),
    ],
)
def test_delivery_fee_valid_weights(weight, expected):
    assert delivery_fee(weight) == expected


@pytest.mark.parametrize("weight", [-1, 0, 30.1])
def test_delivery_fee_invalid_weights(weight):
    with pytest.raises(ValueError):
        delivery_fee(weight)


def test_average_fee():
    assert average_fee([1, 5, 20]) == pytest.approx(113.333333)


def test_average_fee_empty_list():
    with pytest.raises(ValueError):
        average_fee([])
```
