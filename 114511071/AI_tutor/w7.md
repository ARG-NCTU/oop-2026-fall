### 5. Verification

**My final C++ extension:**

```cpp
#include <pybind11/pybind11.h>
#include <pybind11/stl.h>

#include <cstddef>
#include <stdexcept>
#include <vector>

namespace py = pybind11;

double average_temperature(const std::vector<double>& readings) {
    if (readings.empty()) {
        throw std::invalid_argument("readings cannot be empty");
    }

    double total = 0.0;

    for (double value : readings) {
        total += value;
    }

    return total / readings.size();
}

class TemperatureLog {
public:
    void add(double value) {
        readings_.push_back(value);
    }

    std::size_t count() const {
        return readings_.size();
    }

    double average() const {
        return average_temperature(readings_);
    }

private:
    std::vector<double> readings_;
};

PYBIND11_MODULE(tempcpp, m) {
    m.def("average_temperature", &average_temperature);

    py::class_<TemperatureLog>(m, "TemperatureLog")
        .def(py::init<>())
        .def("add", &TemperatureLog::add)
        .def("count", &TemperatureLog::count)
        .def("average", &TemperatureLog::average);
}
```

**My Python verification code:**
