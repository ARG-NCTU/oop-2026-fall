### 5. Verification

**My final program:**

```cpp
#include <iostream>

class TemperatureTracker {
public:
    TemperatureTracker(int capacity)
        : m_data(new double[capacity]),
          m_capacity(capacity),
          m_count(0) {
        for (int i = 0; i < m_capacity; ++i) {
            m_data[i] = 0.0;
        }
    }

    ~TemperatureTracker() {
        delete[] m_data;
    }

    bool add(double value) {
        if (value < -50.0 || value > 100.0) {
            return false;
        }

        if (m_count >= m_capacity) {
            return false;
        }

        m_data[m_count] = value;
        ++m_count;
        return true;
    }

    int count() const {
        return m_count;
    }

    double average() const {
        if (m_count == 0) {
            return 0.0;
        }

        double total = 0.0;

        for (int i = 0; i < m_count; ++i) {
            total += m_data[i];
        }

        return total / m_count;
    }

private:
    double *m_data;
    int m_capacity;
    int m_count;
};

int main() {
    TemperatureTracker tracker(3);

    tracker.add(20.0);
    tracker.add(150.0);
    tracker.add(40.0);

    std::cout << "count = " << tracker.count() << '\n';
    std::cout << "average = " << tracker.average() << '\n';

    return 0;
}
```
