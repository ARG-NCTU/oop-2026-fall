### Square Patrol Controller

**Problem statement:**

Create a program that simulates a vehicle patrolling four waypoints forming a square. The vehicle begins at the first waypoint and visits the remaining points in order. After reaching the fourth point, it returns to the first point and repeats the route.

The controller processes two commands:

- `MUCHSLO` changes the speed to `1.0 m/s`.
- `MUCHFST` changes the speed to `4.0 m/s`.

**Constraints:**

- The route contains exactly four waypoints.
- Each coordinate is between `-1000` and `1000`.
- `1 <= n <= 100`.
- Initial speed is `1.0 m/s`.

### 5. Verification

**My final program:**

```cpp
#include <iostream>
#include <string>
#include <iomanip>

struct Waypoint {
    double x;
    double y;
};

class PatrolController {
public:
    PatrolController(const Waypoint points[], int size)
        : m_points(new Waypoint[size]),
          m_size(size),
          m_nextIndex(1),
          m_speed(1.0) {
        for (int i = 0; i < m_size; ++i) {
            m_points[i] = points[i];
        }
    }

    ~PatrolController() {
        delete[] m_points;
    }

    bool processCommand(const std::string &command) {
        if (command == "MUCHSLO") {
            m_speed = 1.0;
        } else if (command == "MUCHFST") {
            m_speed = 4.0;
        } else {
            return false;
        }

        std::cout << std::fixed << std::setprecision(1)
                  << "Speed: " << m_speed
                  << ", Next waypoint: ("
                  << m_points[m_nextIndex].x << ", "
                  << m_points[m_nextIndex].y << ")\n";

        m_nextIndex = (m_nextIndex + 1) % m_size;
        return true;
    }

private:
    Waypoint *m_points;
    int m_size;
    int m_nextIndex;
    double m_speed;
};

int main() {
    Waypoint square[4] = {
        {0, 0},
        {0, 10},
        {10, 10},
        {10, 0}
    };

    PatrolController controller(square, 4);

    int numberOfCommands;
    std::cin >> numberOfCommands;

    for (int i = 0; i < numberOfCommands; ++i) {
        std::string command;
        std::cin >> command;

        if (!controller.processCommand(command)) {
            std::cout << "Invalid command\n";
        }
    }

    return 0;
}
```
