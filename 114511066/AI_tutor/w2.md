def most_frequent(nums):
    freqs = {}

    for n in nums:
        if n in freqs:
            freqs[n] += 1
        else:
            freqs[n] = 1

    highest = max(freqs.values())
    result = []

    for n in freqs:
        if freqs[n] == highest:
            result.append(n)

    return (result, highest)
