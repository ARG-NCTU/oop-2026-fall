def safe_average(values):
    try:
        nums = []
        for value in values:
            nums.append(float(value))
        return sum(nums) / len(nums)
    except ZeroDivisionError:
        return 0.0
    except (ValueError, TypeError):
        raise ValueError("invalid value")
